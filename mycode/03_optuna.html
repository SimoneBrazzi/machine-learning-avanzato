1. Theory

Evolution of the Bayesian Search, which exclude a priori part of the hyperparameter space BEFORE the fit. It is particularly useful with big hyperparameter spaces. It uses a Tree structured parzen estimator cycle: - Objective function: function that takes hyperparameters as input and return a metric to minimize/maximize. - Define a parameter distribution. - Likelihood estimation: two probability density functions are built, one for good trials and one for bad trials. - Probabilistic model based on the 2 probability density functions. One is cutted off if the result is a low performance. - Future trials are selected by maximizing the Expected Improvement.

\[ P(y | x, \theta) \]

  • Dynamic search in a binded hyperparameter space.
  • Pruning of not promising trials during the training.
    • PROS: curvesomputationally efficient.
    • PROS: Flexilbility.
    • CONS: pruning too strong with simple problems.
    • CONS: Steeper learning curves.
  • sklearn approach and ease of integration with other libraries.

import matplotlib.pyplot as plt
import optuna

from sklearn.metrics import accuracy_score
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.svm import SVC 
import numpy as np 

def optuna_classification(x, y, cv, n_trials):

  # n_trials = max number of trials so there is an upper limit to exit the study

  xtrain, xtest, ytrain, ytest = train_test_split(x, y, test_size=0.2, random_state=42)

  def objective(trial):

    C = trial.suggest_float("C", 1e-10, 1e+10, log=True)
    gamma = trial.suggest_categorical('gamma', ['scale', 'auto'])
    kernel = trial.suggest_categorical('kernel', ['linear', 'poly', 'rbf', 'sigmoid'])

    model = SVC(C=C, gamma=gamma, kernel=kernel, probability=True, random_state=42)
    score = cross_val_score(model, xtrain, ytrain, cv=cv, scoring='accuracy').mean()

    return score

  study = optuna.create_study(direction='maximize', study_name='study 1', sampler=optuna.samplers.TPESampler(seed=42))
  study.optimize(objective, n_trials=n_trials)
  best_params = study.best_params
  best_model = SVC(C=best_params['C'], gamma=best_params['gamma'], kernel=best_params['kernel'], probability=True, random_state=42)
  best_model.fit(xtrain, ytrain)

  ytrain_pred = best_model.predict(xtrain)
  ytest_pred = best_model.predict(xtest)
  train_accuracy = accuracy_score(ytrain, ytrain_pred)
  test_accuracy = accuracy_score(ytest, ytest_pred)

  print(f"best params: {best_params}")
  print(f"train accuracy: {train_accuracy:.4f}")
  print(f"test accuracy: {test_accuracy:.4f}")

  study_accuracies = [trial.value for trial in study.trials]
  plt.plot(study_accuracies)
  plt.xlabel("Trial Number")
  plt.ylabel("Accuracy")
  plt.title("Accuracy by Trial Number")
  plt.show()

  return best_model

def make_grid_search(x, y, cv, grid):
  tic = time()

  svr = SVR()
  grid_search = GridSearchCV(
    estimator=svr,
    param_grid=grid,
    scoring='r2',
    n_jobs=-1,
    cv=cv
  )

  grid_search.fit(xtrain, ytrain)

  best_params = grid_search.best_params_
  best_model = grid_search.best_estimator_
  best_score = grid_search.best_score_

  ytest_pred = grid_search.predict(xtest)
  r2_test = r2_score(ytest, ytest_pred)

  print(f"Best params: {best_params}.")
  print(f"Best model: {best_model}.")
  print(f"Best R2-score: {r2_test}.")
  print(f"Time elapsed: {time() - tic :.4f} seconds.")

  return

def make_optuna(cv, n_trials):

  tic = time()

  def objective(trial):

    C = trial.suggest_float("C", 1e-10, 1e+10, log=True)
    gamma = trial.suggest_categorical('gamma', ['scale', 'auto'])
    kernel = trial.suggest_categorical('kernel', ['linear', 'poly', 'rbf', 'sigmoid'])

    svr = SVR(C=C, gamma=gamma, kernel=kernel, max_iter=10000)
    score = cross_val_score(
      estimator=svr,
      X=xtrain,
      y=ytrain,
      scoring='r2',
      cv=cv
    ).mean()

    return score

  study = optuna.create_study(
    direction="maximize",
    study_name= "study_comparison",
    sampler=optuna.samplers.TPESampler(seed=42)
  )
  study.optimize(objective, n_trials=n_trials)

  best_params = study.best_params
  best_model = SVR(
    C=best_params["C"],
    gamma=best_params["gamma"],
    kernel=best_params["kernel"]
  )
  best_model.fit(xtrain, ytrain)


  ytrain_pred = best_model.predict(xtrain)
  ytest_pred = best_model.predict(xtest)
  train_accuracy = r2_score(ytrain, ytrain_pred)
  test_accuracy = r2_score(ytest, ytest_pred)

  print(f"best params: {best_params}")
  print(f"train accuracy: {train_accuracy:.4f}")
  print(f"test accuracy: {test_accuracy:.4f}")
  print(f"Elapsed time: {time() - tic :.4f}.")

  study_scores = [trial.value for trial in study.trials]
  plt.plot(study_scores)
  plt.xlabel("Trial Number")
  plt.ylabel("R2 score")
  plt.title("R2 Score by Trial Number")
  plt.show()

  return best_model
